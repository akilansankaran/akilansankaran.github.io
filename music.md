<script>
function toggleRepertoire(box) {
  box.classList.toggle('expanded');
}

/* Auto-fill each era header with a live piece count */
document.querySelectorAll('.era-card').forEach(function (card) {
  var count = card.querySelectorAll('.piece-list > li').length;
  var badge = document.createElement('span');
  badge.className = 'era-count';
  badge.textContent = count + (count === 1 ? ' work' : ' works');
  card.querySelector('.era-head-main').appendChild(badge);
});
</script>
